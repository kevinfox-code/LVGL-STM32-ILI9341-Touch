cmake_minimum_required(VERSION 3.20)
project(firmware_host_tests C)
enable_testing()
option(SANITIZE "Enable address and undefined behavior sanitizers" ON)
foreach(VARIANT bme680 pwm-slider squareline-base lvgl-start)
    set(V "${CMAKE_CURRENT_SOURCE_DIR}/../variants/${VARIANT}")
    add_executable(test-${VARIANT} test_drivers.c
        "${V}/Core/Src/ILI9341.c" "${V}/Core/Src/XPT2046.c"
        "${V}/Core/Src/TouchCalibration.c" "${V}/Core/Src/LCDController.c")
    target_include_directories(test-${VARIANT} PRIVATE mocks "${V}/Core/Inc")
    target_compile_options(test-${VARIANT} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    if(VARIANT STREQUAL "pwm-slider")
        target_sources(test-${VARIANT} PRIVATE "${V}/Drivers/ui/ui_events.c")
        target_compile_definitions(test-${VARIANT} PRIVATE _PWM_SLIDER_UI_H TEST_SLIDER)
    endif()
    if(SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(test-${VARIANT} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(test-${VARIANT} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME ${VARIANT}-regressions COMMAND test-${VARIANT})
endforeach()
add_executable(test-bme680-sensor test_bme680.c ../variants/bme680/Core/Src/bme68xController.c)
target_include_directories(test-bme680-sensor PRIVATE mocks ../variants/bme680/Core/Inc ../variants/bme680/Drivers/BME68x)
target_compile_options(test-bme680-sensor PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test-bme680-sensor PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test-bme680-sensor PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME bme680-sensor-regressions COMMAND test-bme680-sensor)
